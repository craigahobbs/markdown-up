/**
 * The MarkdownUp application
 */
export class MarkdownUp {
    /**
     * Create an application instance
     *
     * @param {Object} window - The web browser window object
     * @param {Object} [options] - The [application options]{@link module:lib/app~MarkdownUpOptions}
     */
    constructor(window, options = null) {
        this.window = window;
        this.params = null;
        this.paramsLocal = null;
        this.paramsSession = null;
        this.codeBlocks = (options !== null ? options.codeBlocks : null) ?? null;
        this.darkMode = (options !== null ? options.darkMode : null) ??
            (this.window.matchMedia && this.window.matchMedia('(prefers-color-scheme: dark)').matches);
        this.fontSize = (options !== null ? options.fontSize : null) ?? 12;
        this.globals = (options !== null ? options.globals : null) ?? null;
        this.hashURL = (options !== null ? options.hashURL : null) ?? 'sameOrigin';
        this.helpURL = (options !== null ? options.helpURL : null) ?? defaultHelpURL;
        this.lineHeight = (options !== null ? options.lineHeight : null) ?? 1.3;
        this.markdownText = (options !== null ? options.markdownText : null) ?? null;
        this.menu = (options !== null ? options.menu : null) ?? true;
        this.url = (options !== null ? options.url : null) ?? 'README.md';
        this.runtimeDocumentKeyDown = null;
        this.runtimeWindowResize = null;
        this.runtimeTimeoutId = null;
        this.runtimeCount = 0;

        // The page's content element model, as last rendered (for reading the page aloud)
        this.pageElements = null;

        // The voice playback. It's either the page read aloud (the menu's "Read aloud") or a script's (voiceSpeak) -
        // voiceScript is the script's Markdown text and final arguments.
        this.voice = new MarkdownUpVoice(window, () => this.renderVoice(true));
        this.voiceReadAloud = false;
        this.voiceScript = null;

        // The voice playback bar's own element, if any (the page has no voice playback bar element)
        this.voiceBar = null;

        // The shared Web Audio state (for windowPlaySound), created on first use. The noise buffer is
        // a one-second white-noise buffer reused for drum and explosion sounds.
        this.audioContext = null;
        this.audioNoiseBuffer = null;

        // Track the live keyboard state (for windowKeyState polling). Keys are physical key codes
        // (e.g. "ArrowUp"); "ctrl"/"shift"/"alt"/"meta" hold the latest modifier state.
        this.keyState = {};
        this.window.addEventListener('keydown', (event) => this.updateKeyState(event, true));
        this.window.addEventListener('keyup', (event) => this.updateKeyState(event, false));
        this.window.addEventListener('blur', () => {
            for (const code of Object.keys(this.keyState)) {
                this.keyState[code] = false;
            }
        });

        // The escape key closes the menu
        this.window.addEventListener('keydown', (event) => {
            if (event.key === 'Escape' && this.menu && this.paramsSession !== null && 'menu' in this.paramsSession) {
                this.menuSessionToggle('menu');
            }
        });
    }


    // Record a keyboard key's down/up state along with the current modifier flags
    updateKeyState(event, down) {
        this.keyState[event.code] = down;
        this.keyState.ctrl = event.ctrlKey;
        this.keyState.shift = event.shiftKey;
        this.keyState.alt = event.altKey;
        this.keyState.meta = event.metaKey;
    }


    // Get the shared Web Audio context, creating it on first use
    getAudioContext() {
        if (this.audioContext === null) {
            this.audioContext = new this.window.AudioContext();
        }
        return this.audioContext;
    }


    // Get the shared one-second white-noise buffer (for windowPlaySound drums), creating it on first use
    getAudioNoiseBuffer() {
        const audioContext = this.getAudioContext();
        if (this.audioNoiseBuffer === null) {
            const {sampleRate} = audioContext;
            this.audioNoiseBuffer = audioContext.createBuffer(1, sampleRate, sampleRate);
            const data = this.audioNoiseBuffer.getChannelData(0);
            for (let ix = 0; ix < data.length; ix++) {
                data[ix] = Math.random() * 2 - 1;
            }
        }
        return this.audioNoiseBuffer;
    }


    /**
     * Run the application
     */
    async run() {
        await this.render();
        this.window.addEventListener('hashchange', () => this.render(), false);
    }


    async render(forceRender = false) {
        // Parse the hash parameters and render the application element model
        let result;
        let isError = false;
        const urlPrev = this.params !== null ? (this.params.url ?? null) : null;
        try {
            // Validate hash parameters
            const paramsPrev = this.params;
            this.updateParams();

            // Skip the render if the page params haven't changed
            if (!forceRender && paramsPrev !== null && valueJSON(paramsPrev) === valueJSON(this.params)) {
                return;
            }
        } catch ({message}) {
            result = {
                'title': 'MarkdownUp',
                'elements': {'html': 'p', 'elem': {'text': `Error: ${message}`}}
            };
            isError = true;
        }

        // A new page ends the reading of the page aloud, a new document ends a script's voice playback, and rendering
        // the page anew ends its voice lines
        const url = this.params !== null ? (this.params.url ?? null) : null;
        if (!forceRender && (this.voiceReadAloud || url !== urlPrev)) {
            this.voice.close();
        }
        this.voice.sayStop();

        // Set the colors
        const isDarkMode = this.paramsLocal.darkMode ?? this.darkMode;
        this.window.document.documentElement.style.colorScheme = isDarkMode ? 'dark' : 'light';

        // Set the font size
        const fontSize = this.paramsLocal.fontSize ?? this.fontSize;
        this.window.document.documentElement.style.setProperty('--barescript-font-size', `${fontSize}pt`);

        // Set the line height
        const lineHeight = this.paramsLocal.lineHeight ?? this.lineHeight;
        this.window.document.documentElement.style.setProperty('--barescript-line-height', `${lineHeight}em`);

        // Call the application main and validate the result
        this.runtimeCount += 1;
        this.clearRuntimeCallbacks();
        this.pageElements = null;
        if (!isError) {
            result = await this.main();
        }

        // Set the window title
        const title = result.title ?? null;
        if (title !== null) {
            this.window.document.title = title;
        }

        // Timeout?
        if ('timeout' in result) {
            this.runtimeTimeoutId = this.window.setTimeout(...result.timeout);
        }

        // Window resize event handler?
        if ('resize' in result) {
            this.runtimeWindowResize = result.resize;
            this.window.addEventListener('resize', result.resize);
        }

        // Set the document keydown handler? (clearRuntimeCallbacks removes it)
        if ('keyDown' in result) {
            this.runtimeDocumentKeyDown = async (event) => {
                // Is the active element something that wants this event?
                const {activeElement} = this.window.document;
                if (activeElement instanceof this.window.HTMLElement &&
                    (activeElement.tagName === 'INPUT' ||
                     activeElement.tagName === 'TEXTAREA' ||
                     activeElement.tagName === 'SELECT' ||
                     activeElement.isContentEditable ||
                     activeElement.getAttribute('role') === 'textbox' ||
                     activeElement.getAttribute('role') === 'combobox' ||
                     activeElement.getAttribute('role') === 'searchbox'
                    )
                   ) {
                    return;
                }

                // Call the document keydown handler
                await result.keyDown(event);
            };
            this.window.document.addEventListener('keydown', this.runtimeDocumentKeyDown);
        }

        // Render the element model
        this.renderPage(result.elements);
        this.renderVoice();

        // Navigate?
        // Note: This is done after render since it may have no effect (in which case we need to render)
        if ('location' in result && result.location !== null) {
            this.window.location.href = result.location;

        // If there is a header ID, re-navigate to scroll to the header ID since it was just rendered.
        // The re-render is short-circuited by the unchanged hash param check above.
        } else if (!isError && getHeaderId(this.window.location.hash) !== null) {
            this.window.location.href = this.window.location.hash;
        }

        // Focus?
        if ('focus' in result) {
            this.setDocumentFocus(result.focus);
        }
    }


    updateParams(paramString = null, localJSONString = null, sessionJSONString = null) {
        // Clear, then validate the hash parameters (may throw)
        this.params = null;
        this.paramsLocal = {};
        this.paramsSession = {};

        // Decode and validate the hash parameters
        this.params = schemaValidate(
            markdownUpTypes,
            'MarkdownUp',
            urlDecodeQueryString(paramString ?? this.window.location.hash.slice(1))
        );

        // Decode and validate the local storage paramters
        const localJSON = localJSONString ?? this.window.localStorage.getItem('MarkdownUp');
        if (localJSON !== null) {
            try {
                this.paramsLocal = schemaValidate(markdownUpTypes, 'MarkdownUpLocal', JSON.parse(localJSON));
            } catch {
                // Do nothing
            }
        }

        // Decode and validate the session storage parameters
        const sessionJSON = sessionJSONString ?? this.window.sessionStorage.getItem('MarkdownUp');
        if (sessionJSON !== null) {
            try {
                this.paramsSession = schemaValidate(markdownUpTypes, 'MarkdownUpSession', JSON.parse(sessionJSON));
            } catch {
                // Do nothing
            }
        }
    }


    clearRuntimeCallbacks(clearResize = true) {
        // Clear the runtime timeout ID, if one is set
        if (this.runtimeTimeoutId !== null) {
            this.window.clearTimeout(this.runtimeTimeoutId);
            this.runtimeTimeoutId = null;
        }

        // Clear the window resize event handler
        if (clearResize && this.runtimeWindowResize !== null) {
            this.window.removeEventListener('resize', this.runtimeWindowResize);
            this.runtimeWindowResize = null;
        }

        // Clear the document keydown event handler
        if (clearResize && this.runtimeDocumentKeyDown !== null) {
            this.window.document.removeEventListener('keydown', this.runtimeDocumentKeyDown);
            this.runtimeDocumentKeyDown = null;
        }
    }


    setDocumentFocus(focusId) {
        const element = this.window.document.getElementById(focusId);
        if (element !== null) {
            element.focus();
            if ('selectionStart' in element) {
                element.selectionStart = element.value.length;
                element.selectionEnd = element.value.length;
            }
        }
    }


    async main() {
        const url = this.getResourceURL();
        const isHashURL = 'url' in this.params && this.params.url !== '';
        const scriptOptions = this.createScriptOptions();

        // Get the Markdown text
        let markdownText;
        let timeBegin;
        if (this.markdownText !== null && !isHashURL) {
            ({markdownText} = this);

            // Log Markdown render begin
            if (scriptOptions.debug) {
                timeBegin = performance.now();
                scriptOptions.logFn('MarkdownUp: ===== Rendering Markdown text');
            }
        } else {
            // Is the "url" hash parameter's resource URL allowed?
            if (isHashURL && !this.isHashURLAllowed(url)) {
                return hashURLError(url);
            }

            // Log Markdown render begin
            if (scriptOptions.debug) {
                scriptOptions.logFn(`MarkdownUp: ===== Rendering Markdown document "${url}"`);
            }

            // Log Markdown fetch begin
            let fetchBegin;
            if (scriptOptions.debug) {
                fetchBegin = performance.now();
                scriptOptions.logFn(`MarkdownUp: Fetching "${url}" ...`);
            }

            // Fetch the Markdown text resource URL
            const response = await this.window.fetch(url);

            // Was the "url" hash parameter's resource URL redirected to a resource URL that's not allowed?
            const responseURL = response.url ?? '';
            if (isHashURL && responseURL !== '' && !this.isHashURLAllowed(responseURL)) {
                return hashURLError(responseURL);
            }

            if (!response.ok) {
                const status = response.statusText;
                return {
                    'title': 'MarkdownUp',
                    'elements': {
                        'html': 'p',
                        'elem': {'text': `Error: Could not fetch "${url}"${status === '' ? '' : ` - ${JSON.stringify(status)}`}`}
                    }
                };
            }
            markdownText = await response.text();

            // Log Markdown fetch end with timing
            if (scriptOptions.debug) {
                const fetchEnd = performance.now();
                timeBegin = performance.now();
                scriptOptions.logFn(`MarkdownUp: Fetch completed in ${(fetchEnd - fetchBegin).toFixed(1)} milliseconds`);
            }
        }

        // Parse the Markdown and get the title
        const markdownModel = markdownParse(markdownText);
        const title = markdownTitle(markdownModel);

        // Display the Markdown?
        if (this.params.view === 'markdown') {
            this.pageElements = markdownElements(markdownModel);
            return {
                'title': title,
                'elements': [
                    this.burgerElements(),
                    {'html': 'div', 'attr': {'class': 'markdown'}, 'elem': {'text': markdownText}}
                ]
            };
        }

        // Copy the markdown options and override the markdown-script code block renderer
        const scriptMarkdownOptions = {...scriptOptions.markdownOptions};
        scriptMarkdownOptions.codeBlocks = {...scriptMarkdownOptions.codeBlocks};
        scriptMarkdownOptions.codeBlocks['markdown-script'] = ([codeBlock]) => markdownScriptCodeBlock(codeBlock, scriptOptions);

        // Render the Markdown
        const contentElements = await markdownElementsAsync(markdownModel, scriptMarkdownOptions);
        elementModelWrapCallbacks(contentElements, scriptOptions);
        this.pageElements = contentElements;
        const result = {
            'title': title,
            'elements': [
                // Menu "burger"
                this.burgerElements(),

                // Render the markdown
                contentElements
            ]
        };

        // Set any runtime side-effects
        if (scriptOptions.runtime.documentFocus !== null) {
            result.focus = scriptOptions.runtime.documentFocus;
        }
        if (scriptOptions.runtime.documentKeyDown !== null) {
            result.keyDown = scriptOptions.runtime.documentKeyDown;
        }
        if (scriptOptions.runtime.documentTitle !== null) {
            result.title = scriptOptions.runtime.documentTitle;
        }
        if (scriptOptions.runtime.windowLocation !== null) {
            result.location = scriptOptions.runtime.windowLocation;
        }
        if (scriptOptions.runtime.windowResize !== null) {
            result.resize = scriptOptions.runtime.windowResize;
        }
        if (scriptOptions.runtime.windowTimeout !== null) {
            result.timeout = scriptOptions.runtime.windowTimeout;
        }

        // Reset the runtime
        scriptOptions.runtime.reset();

        // Log Markdown render end
        if (scriptOptions.debug) {
            const timeEnd = performance.now();
            scriptOptions.logFn(`MarkdownUp: Markdown rendered in ${(timeEnd - timeBegin).toFixed(1)} milliseconds`);
        }

        return result;
    }


    getResourceURL() {
        if ('url' in this.params && this.params.url !== '') {
            return this.params.url;
        }
        return this.url;
    }


    // Is the "url" hash parameter's resource URL allowed? The resource's scripts run with the page's origin access.
    isHashURLAllowed(url) {
        if (this.hashURL === 'any') {
            return true;
        } else if (this.hashURL !== 'sameOrigin') {
            return false;
        }

        // Same-origin? Resolve the URL as fetch does - this also excludes "data:" and "blob:" URLs.
        try {
            const {protocol, host} = new URL(url, this.window.document.baseURI);
            return protocol === this.window.location.protocol && host === this.window.location.host;
        } catch {
            return false;
        }
    }


    createScriptOptions() {
        // Create the markdown options object
        const markdownOptions = {
            'codeBlocks': this.codeBlocks ?? {},
            'copyLinks': true,
            'urlFn': ([url]) => this.modifyURL(url),
            'headerIds': true,
            'usedHeaderIds': {}
        };

        // Create the script options object
        const logFn = (text) => {
            this.window.console.log(text);
        };
        const scriptOptions = {
            'audioContextFn': () => this.getAudioContext(),
            'audioNoiseBufferFn': () => this.getAudioNoiseBuffer(),
            'debug': 'debug' in this.paramsSession,
            // eslint-disable-next-line require-await
            'fetchFn': async (fetchURL, options) => this.window.fetch(fetchURL, options),
            'fontSize': this.paramsLocal.fontSize ?? this.fontSize,
            'keyStateFn': () => this.keyState,
            logFn,
            markdownOptions,
            'params': this.params,
            'urlFn': (url) => this.modifyURL(url),
            'voice': {
                'close': () => this.voiceScriptClose(),
                'load': () => this.voice.prepare(),
                'ready': () => this.voice.ready(),
                'say': (text, voice, style, rate) => this.voice.say(text, voice, style, rate),
                'setText': (text, final) => this.voiceScriptSetText(text, final),
                'speak': (text, final, key) => this.voiceScriptSpeak(text, final, key),
                'state': () => this.voiceScriptState()
            },
            'window': this.window
        };

        // Add hash parameter variables, if any
        if ('var' in this.params) {
            scriptOptions.variables = {};
            for (const varName of Object.keys(this.params.var)) {
                const varExprStr = this.params.var[varName];
                try {
                    scriptOptions.variables[varName] = evaluateExpression(barescriptParseExpression(varExprStr), scriptOptions.variables);
                } catch ({message}) {
                    if (scriptOptions.debug) {
                        logFn(`MarkdownUp: Error evaluating variable "${varName}" expression "${varExprStr}": ${message}`);
                    }
                }
            }
        }

        // Create the markdown-script runtime
        const runtime = new MarkdownScriptRuntime(scriptOptions);
        const {runtimeCount} = this;
        scriptOptions.globals = {...markdownScriptFunctions};
        if (this.globals !== null) {
            Object.assign(scriptOptions.globals, this.globals);
        }
        scriptOptions.runtime = runtime;
        scriptOptions.runtimeUpdateFn = () => {
            // Did a render occur in the middle of this runtime update? If so, just reset the runtime.
            if (this.runtimeCount !== runtimeCount) {
                runtime.reset();
                return;
            }

            // Did script set the document title?
            if (runtime.documentTitle !== null) {
                this.window.document.title = runtime.documentTitle;
            }

            // Did the script set a timeout?
            this.clearRuntimeCallbacks(false);
            if (runtime.windowTimeout !== null) {
                this.runtimeTimeoutId = this.window.setTimeout(...runtime.windowTimeout);
            }

            // Reset the runtime and render
            const elements = runtime.resetElements();
            if (elements !== null) {
                const {body} = this.window.document;
                if (runtime.documentReset === null) {
                    this.renderPage([this.burgerElements(), elements]);
                    this.pageElements = elements;
                } else {
                    while (body.lastChild !== null && body.lastChild !== this.voiceBar && body.lastChild.id !== runtime.documentReset) {
                        body.lastChild.remove();
                    }
                    renderElements(body, [elements], false);
                    this.pageElements = [pageElementsReset(this.pageElements, runtime.documentReset), elements];
                }
                this.renderVoice();
            }

            // Navigate?
            // Note: This is done after render since it may have no effect (in which case we want to render).
            if (runtime.windowLocation !== null) {
                this.window.location.href = runtime.windowLocation;
            }

            // Focus?
            if (runtime.documentFocus !== null) {
                this.setDocumentFocus(runtime.documentFocus);
            }

            // Reset the runtime
            runtime.reset();
        };

        return scriptOptions;
    }


    modifyURL(url) {
        const resourceURL = this.getResourceURL();

        // Hash URL?
        if (url.startsWith('#')) {
            // Fixup the "url" param if its relative
            const hashParams = urlDecodeQueryString(url.slice(1)) ?? {};
            if ('url' in hashParams && hashParams.url !== '' && isRelativeURL(hashParams.url)) {
                hashParams.url = `${getBaseURL(resourceURL)}${hashParams.url}`;
            }

            // Encode hash parameters with a header ID
            const paramString = urlEncodeQueryString({...this.params, ...hashParams});
            const headerId = getHeaderId(url);
            return `#${paramString}${headerId !== null && paramString !== '' ? '&' : ''}${headerId !== null ? headerId : ''}`;
        }

        // Relative URL?
        if (isRelativeURL(url)) {
            return `${getBaseURL(resourceURL)}${url}`;
        }

        return url;
    }


    burgerElements() {
        const paramString = urlEncodeQueryString(this.params);
        const topHeaderId = `${paramString}${paramString !== '' ? '&' : ''}_top`;
        const isMenu = 'menu' in this.paramsSession;
        return [
            !this.menu ? null : [
                // Menu "burger" button
                {
                    'html': 'div',
                    'attr': {'class': 'menu-burger'},
                    'elem': {
                        'html': 'button',
                        'attr': {'type': 'button', 'class': 'menu-button', 'aria-label': 'Menu', 'aria-expanded': `${isMenu}`},
                        'elem': iconElements(icons.burger, 22),
                        'callback': clickCallback(() => this.menuSessionToggle('menu'))
                    }
                },

                // Popup menu - a click outside of it (on its backdrop) closes it
                !isMenu ? null : [
                    {
                        'html': 'div',
                        'attr': {'class': 'menu-backdrop'},
                        'callback': clickCallback(() => this.menuSessionToggle('menu'))
                    },
                    {
                        'html': 'div',
                        'attr': {'class': 'menu', 'role': 'menu'},
                        'elem': [
                            menuItemElements('Read aloud', icons.readAloud, null, () => this.menuReadAloud()),
                            menuSeparatorElements(),
                            menuItemElements(
                                'Dark mode', icons.darkMode, !!(this.paramsLocal.darkMode ?? this.darkMode), () => this.menuDarkModeToggle()
                            ),
                            this.menuStepperElements('Font size', icons.fontSize, 'fontSize', this.fontSize, menuFontSizes, 'pt'),
                            this.menuStepperElements('Line height', icons.lineHeight, 'lineHeight', this.lineHeight, menuLineHeights, ''),
                            menuSeparatorElements(),
                            menuItemElements(
                                'Markdown source', icons.markdown, this.params.view === 'markdown', () => this.menuViewToggle('markdown')
                            ),
                            menuItemElements('Debug', icons.debug, 'debug' in this.paramsSession, () => this.menuSessionToggle('debug')),
                            {
                                'html': 'a',
                                'attr': {'class': 'menu-item', 'role': 'menuitem', 'href': this.helpURL},
                                'elem': [
                                    iconElements(icons.help),
                                    {'html': 'span', 'attr': {'class': 'menu-label'}, 'elem': {'text': 'Help'}},
                                    iconElements(icons.link, 14)
                                ]
                            }
                        ]
                    }
                ]
            ],

            // The top header ID
            {'html': 'div', 'attr': {'id': topHeaderId, 'style': 'display=none; position: absolute; top: 0;'}}
        ];
    }


    // Create a menu stepper's element model - a local storage value with buttons to step it down and up through its values
    menuStepperElements(label, icon, valueName, valueDefault, values, unit) {
        const value = this.paramsLocal[valueName] ?? valueDefault;
        const stepElements = (stepLabel, stepText, valueNew) => ({
            'html': 'button',
            'attr': {
                'type': 'button',
                'class': 'menu-step',
                'aria-label': `${label} ${stepLabel}`,
                'disabled': valueNew === null ? 'disabled' : null
            },
            'elem': {'text': stepText},
            'callback': valueNew === null ? null : clickCallback(() => {
                if (valueNew === valueDefault) {
                    delete this.paramsLocal[valueName];
                } else {
                    this.paramsLocal[valueName] = valueNew;
                }
                this.window.localStorage.setItem('MarkdownUp', JSON.stringify(this.paramsLocal));
                this.render(true);
            })
        });
        return {
            'html': 'div',
            'attr': {'class': 'menu-item menu-stepper'},
            'elem': [
                iconElements(icon),
                {'html': 'span', 'attr': {'class': 'menu-label'}, 'elem': {'text': label}},
                stepElements('down', '−', values.findLast((valueOther) => valueOther < value) ?? null),
                {'html': 'span', 'attr': {'class': 'menu-value'}, 'elem': {'text': `${value}${unit}`}},
                stepElements('up', '+', values.find((valueOther) => valueOther > value) ?? null)
            ]
        };
    }


    // Toggle a session storage value (e.g., the menu) and render
    menuSessionToggle(valueName) {
        if (valueName in this.paramsSession) {
            delete this.paramsSession[valueName];
        } else {
            this.paramsSession[valueName] = 1;
        }
        this.window.sessionStorage.setItem('MarkdownUp', JSON.stringify(this.paramsSession));
        this.render(true);
    }


    // Toggle a view (e.g., the Markdown text view)
    menuViewToggle(viewValue) {
        const params = {...this.params};
        if (params.view === viewValue) {
            delete params.view;
        } else {
            params.view = viewValue;
        }
        this.window.location.href = `#${urlEncodeQueryString(params)}`;
    }


    // Toggle dark mode and render
    menuDarkModeToggle() {
        const isDarkMode = this.paramsLocal.darkMode ?? this.darkMode;
        if (isDarkMode !== this.darkMode) {
            delete this.paramsLocal.darkMode;
        } else {
            this.paramsLocal.darkMode = !isDarkMode;
        }
        this.window.localStorage.setItem('MarkdownUp', JSON.stringify(this.paramsLocal));
        this.render(true);
    }


    // Read the page aloud - the page as it's rendered now, if it has anything to read - and close the menu
    menuReadAloud() {
        const paragraphs = voiceElementsParagraphs(this.pageElements, true);
        if (paragraphs.length !== 0) {
            this.voiceReadAloud = true;
            this.voiceScript = null;
            this.voiceSettings();
            this.voice.speak(paragraphs, true, null);
        }
        this.menuSessionToggle('menu');
    }


    // Set the voice playback's voice and speaking rate to the preferred voice and speaking rate
    voiceSettings() {
        this.voice.voice = this.paramsLocal.voice ?? null;
        this.voice.rate = this.paramsLocal.voiceRate ?? 1;
    }


    // The script library's voiceSpeak - start reading Markdown text aloud
    voiceScriptSpeak(text, final, key) {
        this.voiceReadAloud = false;
        this.voiceScript = [text, final];
        this.voiceSettings();
        this.voice.speak(voiceMarkdownParagraphs(text), final, key);
    }


    // The script library's voiceSetText - set the Markdown text of the script's voice playback, if any
    voiceScriptSetText(text, final) {
        if (this.voiceScriptState() !== null && (text !== this.voiceScript[0] || final !== this.voiceScript[1])) {
            this.voiceScript = [text, final];
            this.voice.setText(voiceMarkdownParagraphs(text), final);
        }
    }


    // The script library's voiceState - the state of the script's voice playback, if any (not the page read aloud)
    voiceScriptState() {
        return this.voiceReadAloud ? null : this.voice.state();
    }


    // The script library's voiceClose - close the script's voice playback, if any (not the page read aloud), and stop
    // the voice lines
    voiceScriptClose() {
        this.voice.sayStop();
        if (!this.voiceReadAloud) {
            this.voice.close();
        }
    }


    // Render the page's element model in the document body. The voice playback bar's own element, if any, is left in
    // place (it's the body's first element) - an element that's replaced as it's clicked gets no click, so the bar's
    // controls would fail while an application renders.
    renderPage(elements) {
        const {body} = this.window.document;
        if (this.voiceBar === null) {
            renderElements(body, elements);
        } else {
            while (body.lastChild !== this.voiceBar) {
                body.lastChild.remove();
            }
            renderElements(body, elements, false);
        }
    }


    // Render the voice playback bar, if there's a voice playback - in the page's voice playback bar element, if it has
    // one, or in the bar's own element at the bottom of the window. Unless the playback changed, a rendered bar is left
    // as it is.
    renderVoice(changed = false) {
        const {document} = this.window;
        const isPlayback = this.voice.state() !== null;
        const pageBar = document.getElementById(voiceBarId);

        // Remove the bar's own element when it isn't needed
        if (this.voiceBar !== null && (pageBar !== null || !isPlayback)) {
            this.voiceBar.remove();
            this.voiceBar = null;
        }

        // No playback? Clear the page's voice playback bar element, if any.
        if (!isPlayback) {
            if (pageBar !== null && pageBar.firstChild !== null) {
                pageBar.replaceChildren();
            }
            return;
        }

        // Create the bar's own element, if necessary - the body's first element, where rendering the page leaves it
        if (pageBar === null && this.voiceBar === null) {
            this.voiceBar = document.createElement('div');
            this.voiceBar.id = voiceBarDefaultId;
            this.voiceBar.className = 'voice-default';
            document.body.prepend(this.voiceBar);
        }

        // Render the bar
        const bar = pageBar ?? this.voiceBar;
        if (changed || bar.firstChild === null) {
            renderElements(bar, this.voiceElements());
        }
    }


    // Create the voice playback bar's element model
    voiceElements() {
        const {voice} = this;
        const {playing} = voice.state();
        const textButtonElements = (label, text, onClick) => ({
            'html': 'button',
            'attr': {'type': 'button', 'class': 'voice-text', 'aria-label': label},
            'elem': {'text': text},
            'callback': clickCallback(onClick)
        });
        return {
            'html': 'div',
            'attr': {'class': 'voice', 'role': 'group', 'aria-label': 'Read aloud'},
            'elem': [
                voiceButtonElements('Rewind', icons.rewind, true, () => voice.rewind()),
                voiceButtonElements('Skip back', icons.skipBack, true, () => voice.skipBack()),
                (playing
                    ? voiceButtonElements('Pause', icons.pause, true, () => voice.pause(), true)
                    : voiceButtonElements('Play', icons.play, true, () => voice.play(), true)),
                voiceButtonElements('Skip forward', icons.skipForward, true, () => voice.skipForward()),
                textButtonElements('Voice', voice.voiceLabel(), () => {
                    const voiceNext = voice.voiceNext();
                    if (voiceNext !== null) {
                        this.paramsLocal.voice = voiceNext;
                        this.window.localStorage.setItem('MarkdownUp', JSON.stringify(this.paramsLocal));
                        voice.setVoice(voiceNext);
                    }
                }),
                {'html': 'span', 'attr': {'class': 'voice-separator'}, 'elem': {'text': '|'}},
                textButtonElements('Speaking rate', `${voice.rate}x`, () => {
                    const rateNext = voice.rateNext();
                    this.paramsLocal.voiceRate = rateNext;
                    this.window.localStorage.setItem('MarkdownUp', JSON.stringify(this.paramsLocal));
                    voice.setRate(rateNext);
                }),
                voiceButtonElements('Close', icons.close, false, () => voice.close())
            ]
        };
    }
}


// Create an element model callback that adds an element's click event handler
function clickCallback(onClick) {
    return (element) => {
        element.addEventListener('click', onClick);
    };
}


// Create an icon's element model - an icon is drawn with lines or filled
function iconElements(path, size = 18, fill = false) {
    return {
        'svg': 'svg',
        'attr': {'width': size, 'height': size, 'viewBox': '0 0 24 24', 'aria-hidden': 'true'},
        'elem': {
            'svg': 'path',
            'attr': {
                'fill': fill ? 'currentColor' : 'none',
                'stroke': fill ? 'none' : 'currentColor',
                'stroke-width': 2,
                'stroke-linecap': 'round',
                'stroke-linejoin': 'round',
                'd': path
            }
        }
    };
}


// Create a menu item's element model - if checked isn't null, the item is a toggle
function menuItemElements(label, icon, checked, onClick) {
    const isToggle = checked !== null;
    return {
        'html': 'button',
        'attr': {
            'type': 'button',
            'class': 'menu-item',
            'role': isToggle ? 'menuitemcheckbox' : 'menuitem',
            'aria-checked': isToggle ? `${checked}` : null
        },
        'elem': [
            iconElements(icon),
            {'html': 'span', 'attr': {'class': 'menu-label'}, 'elem': {'text': label}},
            !isToggle ? null : {'html': 'span', 'attr': {'class': 'menu-switch'}}
        ],
        'callback': clickCallback(onClick)
    };
}


// Create a menu separator's element model
function menuSeparatorElements() {
    return {'html': 'div', 'attr': {'class': 'menu-separator', 'role': 'separator'}};
}


// Create a voice playback bar button's element model
function voiceButtonElements(label, icon, fill, onClick, isPrimary = false) {
    return {
        'html': 'button',
        'attr': {'type': 'button', 'class': `voice-button${isPrimary ? ' voice-primary' : ''}`, 'aria-label': label},
        'elem': iconElements(icon, isPrimary ? 22 : 18, fill),
        'callback': clickCallback(onClick)
    };
}


// Get Markdown text's paragraphs to read aloud
function voiceMarkdownParagraphs(text) {
    return voiceElementsParagraphs(markdownElements(markdownParse(text)));
}


// Get the page elements that remain after a document reset - the top-level elements through the reset ID's element
function pageElementsReset(pageElements, resetId) {
    const elements = [pageElements].flat(Infinity).filter((element) => element !== null);
    const ixReset = elements.findIndex((element) => (element.attr ?? null) !== null && element.attr.id === resetId);
    return elements.slice(0, ixReset + 1);
}


// Create the main result for a "url" hash parameter resource URL that's not allowed
function hashURLError(url) {
    return {
        'title': 'MarkdownUp',
        'elements': {'html': 'p', 'elem': {'text': `Error: The "url" hash parameter resource URL "${url}" is not allowed`}}
    };
}


// Get a URL's header ID
function getHeaderId(url) {
    const matchId = url.match(rHeaderId);
    return matchId !== null ? matchId[1] : null;
}

const rHeaderId = /[#&]([^=]+)$/;


// Test if a URL is relative
function isRelativeURL(url) {
    return !rNotRelativeURL.test(url);
}

const rNotRelativeURL = /^(?:[a-z]+:|\/|\?|#)/;


// Get a URL's base URL
function getBaseURL(url) {
    return url.slice(0, url.lastIndexOf('/') + 1);
}
